# Маппинг полей контрагентов: МойСклад API → БД (таблица pbi_counterparties)
# Формат: поле_api | поле_бд | тип_данных [ | трансформация ]
#
# Способы указания поля API:
#   name                              — простое поле
#   meta.href                         — вложенное поле через точку
#   attributeByName("Имя")            — значение доп. поля по точному name
#   attributeByName("Имя").name       — value.name для customentity
#
# Для meta.href и *.meta.href автоматически извлекается UUID из URL
#
accountId                                         | account_id                   | STRING(255)
actualAddress                                     | actual_address               | STRING(255)
actualAddressFull.city                            | actual_address_city          | STRING(255)
archived                                          | archived                     | BOOLEAN
bonusPoints                                       | bonus_points                 | INTEGER
bonusProgram.meta.href                            | bonus_program                | STRING(255)
code                                              | code                         | STRING(255)
companyType                                       | company_type                 | STRING(255)
created                                           | created                      | DATE
description                                       | description                  | TEXT
discountCardNum                                   | discount_card_number         | STRING(255)
externalCode                                      | external_code                | STRING(255)
group.meta.href                                   | group_href                   | STRING(255)
id                                                | uuid                         | STRING(255)
meta.href                                         | id_href                      | STRING(255)
name                                              | name                         | STRING(255)
owner.meta.href                                   | owner_href                   | STRING(255)
priceType.name                                    | price_type                   | STRING(255)
salesAmount                                       | sales_amount                 | INTEGER
shared                                            | shared                       | BOOLEAN
state.name                                        | state                        | STRING(255)
tags                                              | tags                         | JSON
updated                                           | updated                      | DATE
#
